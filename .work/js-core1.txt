function vehiclePlate(car){return ({'car-1':'3ชฏ3250','car-2':'ฆง8434','car-3':'5กล1034'})[car.id]||car.name;}
function selectVehicle(id){const car=state.vehicles.find(c=>c.id===id);if(!car)return;$('car').value=car.name;syncVehicleSelection();}
function syncVehicleSelection(){const car=state.vehicles.find(c=>c.name===$('car').value);document.querySelectorAll('.car-symbol').forEach(button=>button.setAttribute('aria-pressed',String(button.dataset.carId===car?.id)));}
function setupDropdowns(){const val=$('userName').value;$('userName').innerHTML='<option value="">-- เลือกพยาบาล --</option>'+state.nurses.filter(n=>n.active).map(n=>`<option value="${esc(n.id)}">${esc(n.name)} (${esc(n.dept)})</option>`).join('');$('userName').value=val;}
function login(){const n=state.nurses.find(n=>n.id===$('userName').value&&n.active),car=state.vehicles.find(c=>c.name===$('car').value);if(!n||!car||!$('shift').value)return alert('กรุณาเลือกพยาบาล เวร และรถให้ครบ');const draft=state.drafts.find(d=>d.carId===car.id);if(draft){if(draft.nurseId!==n.id||draft.shift!==$('shift').value)return alert('รถคันนี้มีร่างการตรวจค้างอยู่ กรุณาเลือกผู้ตรวจและเวรเดิมเพื่อทำต่อ');currentUser=clone(draft);checkResults=clone(draft.results);}else{currentUser={id:uid(),carId:car.id,car:car.name,nurseId:n.id,userName:n.name,userId:n.id,shift:$('shift').value,startedAt:new Date().toISOString()};checkResults=state.items.filter(i=>i.carId===car.id&&i.active).map(i=>({...i,status:pending,missing:'',batteryLevel:null}));}
 $('loginPage').classList.add('hidden');$('checkPage').classList.remove('hidden');$('headerTitle').textContent='ตรวจรถ '+vehiclePlate(car);$('loginInfo').textContent=`ผู้ตรวจ: ${n.name} | เวร: ${currentUser.shift} | ${vehiclePlate(car)}`;renderChecklistTable();}
function renderChecklistTable(){ $('itemTable').innerHTML=checkResults.map((i,k)=>`<tr><td>${k+1}</td><td>${esc(i.name)}</td><td>${esc(i.standard)}</td><td><input aria-label="วันหมดอายุ ${esc(i.name)}" type="text" value="${esc(i.expire)}" onchange="changeExpirationRow(${k},this.value)" placeholder="วัน/เดือน/ปี" style="width:125px"></td><td class="${i.status==='Checked'?'status-normal':i.status===missing?'status-missing':''}">${esc(i.status)}</td><td>${esc(i.missing||'-')}</td><td>${i.isBattery?`<div class="battery-container">${[0,1,2,3,4].map(v=>`<button class="small-btn" aria-label="แบตเตอรี่ ${v}/4" onclick="clickBatteryUI(${k},${v})" style="background:${i.batteryLevel===v?'#0f766e':'#64748b'}">${v}/4</button>`).join('')}</div>`:`<button class="small-btn" onclick="setItemStatus(${k},'Checked')">✓</button><button class="small-btn warning" onclick="triggerManualValue(${k},'${used}')">ใช้ไป</button>`}<button class="small-btn danger" onclick="triggerManualValue(${k},'${missing}')">ขาด</button></td></tr>`).join('');}
async function saveDraft(nextResults){const saved=await mutate(s=>{s.drafts=s.drafts.filter(d=>d.carId!==currentUser.carId);s.drafts.push({...clone(currentUser),results:nextResults});});if(saved){checkResults=nextResults;renderChecklistTable();}return saved;}
async function changeExpirationRow(k,v){const next=clone(checkResults);next[k].expire=v;await saveDraft(next);}
async function setItemStatus(k,v){const next=clone(checkResults);next[k].status=v;next[k].missing='';await saveDraft(next);}
async function triggerManualValue(k,v){const value=prompt('ระบุจำนวน / รายละเอียด');if(!value?.trim())return;const next=clone(checkResults);next[k].status=v;next[k].missing=value.trim();await saveDraft(next);}
async function clickBatteryUI(k,v){const next=clone(checkResults);Object.assign(next[k],{batteryLevel:v,status:'Checked',missing:v+'/4 ขีด'});await saveDraft(next);}
async function checkAll(){if(!confirm('ยืนยันเฉพาะรายการที่ยังไม่ตรวจว่าได้ตรวจครบแล้ว? รายการขาด/ใช้ไปและแบตเตอรี่จะคงเดิม'))return;await saveDraft(checkResults.map(i=>i.status===pending&&!i.isBattery?{...i,status:'Checked'}:i));}
