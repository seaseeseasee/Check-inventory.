function mergeChanges(base,next,remote){const merged=clone(remote);for(const table of stores){const before=new Map(base[table].map(r=>[r.id,r])),after=new Map(next[table].map(r=>[r.id,r])),latest=new Map(remote[table].map(r=>[r.id,r]));for(const id of new Set([...before.keys(),...after.keys()])){const old=before.get(id),value=after.get(id);if(JSON.stringify(old)===JSON.stringify(value))continue;const current=latest.get(id);if(JSON.stringify(current)!==JSON.stringify(old)&&JSON.stringify(current)!==JSON.stringify(value))throw Error('รายการเดียวกันถูกแก้ไขจากอีกเครื่อง กรุณาโหลดหน้าใหม่เพื่อตรวจสอบ');if(value===undefined)latest.delete(id);else latest.set(id,value);}merged[table]=[...latest.values()];}return merged;}
